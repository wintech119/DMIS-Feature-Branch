Compute overall totals for the summary cards:

SELECT
  COALESCE(SUM(CASE WHEN t.ttype = 'R' THEN t.qty END), 0) AS total_received,
  COALESCE(SUM(CASE WHEN t.ttype = 'I' THEN t.qty END), 0) AS total_issued,
  COALESCE(SUM(CASE WHEN t.ttype = 'R' THEN t.qty END), 0)

FROM public.transaction t

FROM public.transaction t
JOIN public.item      i ON t.item_id = i.item_id
JOIN public.warehouse w ON t.warehouse_id = w.warehouse_id
